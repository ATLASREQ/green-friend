    ⚙️ Для владельца сайта — настройка один раз

    Вы настраиваете всё один раз у себя — дальше пользователи только регистрируются и жмут «Подключить Telegram». О Supabase они даже не узнают.

    1. Создайте одного общего бота: @BotFather → /newbot → скопируйте токен.
    2. Создайте проект на supabase.com, выполните SQL-скрипт ниже; в Authentication → Providers → Email отключите «Confirm email».
    3. Задайте секреты и задеплойте обе Edge Functions (tg-webhook и reminders) — код ниже.
    4. Подключите вебхук командой curl (ниже) — бот начнёт принимать /start от пользователей.
    5. Создайте cron-задачу для функции reminders — она шлёт напоминания, даже когда вкладка закрыта.
    6. Впишите url, anon key и имя бота в OWNER_CONFIG вверху index.html и выкладывайте сайт. Эта панель после этого исчезнет у всех. Поля выше — чтобы протестировать без правки кода.
    🔒 Безопасность: anon key публичный — доступ к данным ограничивает Row Level Security (каждый видит только своё). Токен бота и service_role ключ живут только в секретах сервера и в браузер не попадают.

create table if not exists public.plants (
  id uuid primary key,
  user_id uuid not null references auth.users(id) on delete cascade,
  sid text not null,
  name text, emoji text, nickname text, room text,
  water_days int, feed_days int, repot_months int,
  light text, humidity text, temp text, tip text, toxic boolean,
  last_water timestamptz, last_feed timestamptz, last_repot timestamptz,
  last_reminded_at timestamptz,
  notified jsonb default '{}'::jsonb,
  created_at timestamptz default now()
);
-- Свои карточки видов
create table if not exists public.custom_species (
  id uuid primary key,
  user_id uuid not null references auth.users(id) on delete cascade,
  name text, emoji text,
  water_days int, feed_days int, repot_months int,
  light text, humidity text, temp text, tip text, toxic boolean,
  created_at timestamptz default now()
);
-- Профиль: привязка Telegram-чата
create table if not exists public.profiles (
  id uuid primary key references auth.users(id) on delete cascade,
  chat_id text,
  tg_connected_at timestamptz,
  created_at timestamptz default now()
);
alter table public.plants enable row level security;
alter table public.custom_species enable row level security;
alter table public.profiles enable row level security;
create policy "plants own" on public.plants for all using (auth.uid()=user_id) with check (auth.uid()=user_id);
create policy "custom own" on public.custom_species for all using (auth.uid()=user_id) with check (auth.uid()=user_id);
create policy "profiles own" on public.profiles for all using (auth.uid()=id) with check (auth.uid()=id);
    🤖 Edge Function «tg-webhook» — принимает /start
    // supabase/functions/tg-webhook/index.ts
// Принимает сообщения общего бота Green Friend
import { createClient } from "npm:@supabase/supabase-js@2";

const sb = createClient(
  Deno.env.get("SUPABASE_URL")!,
  Deno.env.get("SUPABASE_SERVICE_ROLE_KEY")!
);
const BOT = Deno.env.get("TELEGRAM_BOT_TOKEN")!;

async function send(chat_id: string, text: string) {
  await fetch(`https://api.telegram.org/bot${BOT}/sendMessage`, {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify({ chat_id, text }),
  });
}

Deno.serve(async (req) => {
  const secret = Deno.env.get("TELEGRAM_SECRET_TOKEN");
  if (secret && req.headers.get("x-telegram-bot-api-secret-token") !== secret) {
    return new Response("forbidden", { status: 403 });
  }
  let update: any;
  try { update = await req.json(); } catch { return new Response("ok"); }
  const msg = update.message;
  if (!msg?.text) return new Response("ok");

  const chatId = String(msg.chat.id);
  const text = msg.text.trim();

  // /start USER_ID — пришёл по ссылке с сайта: привязываем чат
  const m = text.match(/^\/start(?:@\w*)?\s+([\w-]+)/);
  if (m) {
    await sb.from("profiles").upsert({
      id: m[1], chat_id: chatId,
      tg_connected_at: new Date().toISOString(),
    });
    await send(chatId,
      "✅ Готово! Вы подключили напоминания Green Friend.\n\n" +
      "Я напишу, когда растение действительно ждёт: полить 💧, подкормить 🧪 или пересадить 🪴.\n\n" +
      "/test — посмотреть пример напоминания");
  } else if (text.startsWith("/test")) {
    await send(chatId,
      "🧪 Пример напоминания:\n\n🌿 Green Friend напоминает!\n🗓 пятница, 14 августа\n\n" +
      "💧 Полить: 🌿 Монстера · Кухня\n🧪 Подкормить: 🌳 Фикус · Гостиная\n\n" +
      "Хорошего дня вашим зелёным! 🪴");
  } else if (text.startsWith("/status")) {
    const { data } = await sb.from("profiles").select("id").eq("chat_id", chatId).maybeSingle();
    await send(chatId, data
      ? "✅ Чат привязан к аккаунту. Напоминания приходят по расписанию."
      : "⚠️ Чат не привязан. Нажмите «Start» по ссылке с сайта Green Friend.");
  } else {
    await send(chatId, "Я бот Green Friend 🌿\n\n/test — пример напоминания\n/status — проверить привязку\n\nПодключение — по ссылке с сайта.");
  }
  return new Response("ok");
});
    ⏰ Edge Function «reminders» — cron-напоминания
    // supabase/functions/reminders/index.ts
// Запускается cron-задачей и шлёт накопившиеся задачи в Telegram
import { createClient } from "npm:@supabase/supabase-js@2";

const sb = createClient(
  Deno.env.get("SUPABASE_URL")!,
  Deno.env.get("SUPABASE_SERVICE_ROLE_KEY")!
);
const BOT = Deno.env.get("TELEGRAM_BOT_TOKEN")!;

function json(o: any, s = 200) {
  return new Response(JSON.stringify(o), {
    status: s, headers: { "Content-Type": "application/json" },
  });
}

Deno.serve(async () => {
  const { data: plants, error } = await sb.from("plants").select("*");
  if (error || !plants) return json({ error: error?.message }, 500);

  const now = Date.now();
  const COOLDOWN = 12 * 3600e3; // не дёргать чаще двух раз в сутки
  const dueByUser: Record<string, { lines: string[]; ids: string[] }> = {};

  for (const p of plants) {
    const jobs = [
      ["💧 Полить", Date.parse(p.last_water) + (p.water_days ?? 7) * 864e5],
      ["🧪 Подкормить", Date.parse(p.last_feed) + (p.feed_days ?? 14) * 864e5],
      ["🪴 Пересадить", Date.parse(p.last_repot) + (p.repot_months ?? 24) * 2592e6],
    ] as const;
    const due = jobs.filter(([, d]) => d <= now);
    if (!due.length) continue;
    if (now - (p.last_reminded_at ? Date.parse(p.last_reminded_at) : 0) < COOLDOWN) continue;
    const bucket = (dueByUser[p.user_id] ||= { lines: [], ids: [] });
    due.forEach(([label]) => bucket.lines.push(
      `${label}: ${p.emoji} ${p.nickname || p.name}${p.room ? ` · ${p.room}` : ""}`
    ));
    bucket.ids.push(p.id);
  }

  let sent = 0;
  for (const [uid, b] of Object.entries(dueByUser)) {
    const { data: prof } = await sb.from("profiles")
      .select("chat_id").eq("id", uid).maybeSingle();
    if (!prof?.chat_id) continue;
    const text = `🌿 Green Friend напоминает!\n🗓 ${new Date().toLocaleDateString("ru-RU", {
      weekday: "long", day: "numeric", month: "long",
    })}\n\n${b.lines.join("\n")}\n\nХорошего дня вашим зелёным! 🪴`;
    const r = await fetch(`https://api.telegram.org/bot${BOT}/sendMessage`, {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ chat_id: prof.chat_id, text }),
    });
    if (r.ok) {
      sent++;
      await sb.from("plants")
        .update({ last_reminded_at: new Date().toISOString() }).in("id", b.ids);
    }
  }
  return json({ users: Object.keys(dueByUser).length, sent });
});
    🚀 Деплой, вебхук и cron
    # 0. Бот: @BotFather → /newbot → копируем токен (один на всех)

# 1. Supabase CLI (один раз)
npm i -g supabase
supabase login
supabase link --project-ref РЕФ_ПРОЕКТА          # виден в URL проекта

# 2. Секреты
supabase secrets set TELEGRAM_BOT_TOKEN=123456:ABC…
supabase secrets set TELEGRAM_SECRET_TOKEN=любая_длинная_строка

# 3. Деплой функций (папки supabase/functions/tg-webhook и reminders)
supabase functions deploy tg-webhook --no-verify-jwt
supabase functions deploy reminders --no-verify-jwt

# 4. Вебхук (один раз). URL функции смотрите в панели: Edge Functions → tg-webhook
curl "https://api.telegram.org/bot<ТОКЕН>/setWebhook?url=https://<REF>.functions.co/tg-webhook&secret_token=<TELEGRAM_SECRET_TOKEN>"

# 5. Cron: панель Supabase → Cron Jobs → New job
#    тип: Edge Function «reminders», расписание: 0 */2 * * * (каждые 2 часа)
