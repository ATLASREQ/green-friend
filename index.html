Один общий бот — ноль настроек для вас

Не нужно создавать ботов и копировать токены. Просто нажмите «Start» — и напоминания начнут приходить в ваш Telegram.

⚙️ Для владельца сайта — настройка один раз

Вы настраиваете всё один раз у себя — дальше пользователи только регистрируются и жмут «Подключить Telegram». О Supabase они даже не узнают.

  1. Создайте одного общего бота: @BotFather → /newbot → скопируйте токен.
  2. Создайте проект на supabase.com, выполните SQL-скрипт ниже; в Authentication → Providers → Email отключите «Confirm email».
  3. В Authentication → URL Configuration укажите Site URL вашего сайта — иначе не будут работать письма сброса пароля.
  4. Задайте секреты и задеплойте обе Edge Functions (tg-webhook и reminders) — код ниже.
  5. Подключите вебхук (инструкция ниже) — бот начнёт принимать /start от пользователей.
  6. Создайте cron-задачу для функции reminders с расписанием 0 17,19 * * * (20:00 и 22:00 МСК).
  7. Впишите url, anon key и имя бота в OWNER_CONFIG вверху index.html и выкладывайте сайт. Эта панель после этого исчезнет у всех.
🔒 Безопасность: anon key публичный — доступ к данным ограничивает Row Level Security (каждый видит только своё). Токен бота и service_role ключ живут только в секретах сервера и в браузер не попадают.

🤖 Edge Function «tg-webhook» — принимает /start
// supabase/functions/tg-webhook/index.ts
import { createClient } from "npm:@supabase/supabase-js@2";

const sb = createClient(
  Deno.env.get("SUPABASE_URL")!,
  Deno.env.get("SUPABASE_SERVICE_ROLE_KEY")!
);
const BOT = Deno.env.get("TELEGRAM_BOT_TOKEN")!;

async function send(chat_id: string, text: string) {
  await fetch(`https://api.telegram.org/bot${BOT}/sendMessage`, {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify({ chat_id, text }),
  });
}

Deno.serve(async (req) => {
  const secret = Deno.env.get("TELEGRAM_SECRET_TOKEN");
  if (secret && req.headers.get("x-telegram-bot-api-secret-token") !== secret) {
    return new Response("forbidden", { status: 403 });
  }
  let update: any;
  try { update = await req.json(); } catch { return new Response("ok"); }
  const msg = update.message;
  if (!msg?.text) return new Response("ok");

  const chatId = String(msg.chat.id);
  const text = msg.text.trim();

  const m = text.match(/^\/start(?:@\w*)?\s+([\w-]+)/);
  if (m) {
    await sb.from("profiles").upsert({
      id: m[1], chat_id: chatId,
      tg_connected_at: new Date().toISOString(),
    });
    await send(chatId,
      "✅ Готово! Вы подключили напоминания Green Friend.\n\n" +
      "Я напишу, когда растение действительно ждёт: полить 💧, подкормить 🧪 или пересадить 🪴. Напоминания приходят в 20:00 (и повтор в 22:00, если не отметили).\n\n" +
      "/test — посмотреть пример напоминания");
  } else if (text.startsWith("/test")) {
    await send(chatId,
      "🧪 Пример напоминания:\n\n🌿 Green Friend напоминает!\n🗓 среда, 19 августа\n\n" +
      "💧 Полить: 🌿 Монстера · Кухня\n🧪 Подкормить: 🌳 Фикус · Гостиная\n\n" +
      "Хорошего дня вашим зелёным! 🪴");
  } else if (text.startsWith("/status")) {
    const { data } = await sb.from("profiles").select("id").eq("chat_id", chatId).maybeSingle();
    await send(chatId, data
      ? "✅ Чат привязан к аккаунту. Напоминания приходят в 20:00 и 22:00 МСК."
      : "⚠️ Чат не привязан. Нажмите «Start» по ссылке с сайта Green Friend.");
  } else {
    await send(chatId, "Я бот Green Friend 🌿\n\n/test — пример напоминания\n/status — проверить привязку\n\nПодключение — по ссылке с сайта.");
  }
  return new Response("ok");
});
⏰ Edge Function «reminders» — напоминания в 20:00 МСК
// supabase/functions/reminders/index.ts
// Cron (0 17,19 * * * UTC = 20:00 и 22:00 МСК) шлёт напоминания.
// Подкормка — только в сезон роста растения.
import { createClient } from "npm:@supabase/supabase-js@2";

const sb = createClient(
  Deno.env.get("SUPABASE_URL")!,
  Deno.env.get("SUPABASE_SERVICE_ROLE_KEY")!
);
const BOT = Deno.env.get("TELEGRAM_BOT_TOKEN")!;

const OVR: Record<string, [number, number]> = {
  cyclamen: [10, 3], lithops: [10, 3], azalea: [2, 9],
  phalaenopsis: [1, 12], saintpaulia: [1, 12],
};

function json(o: any, s = 200) {
  return new Response(JSON.stringify(o), {
    status: s, headers: { "Content-Type": "application/json" },
  });
}

// Следующий слот напоминания: 17:00/19:00 UTC (20:00/22:00 МСК),
// после 19:00 — следующий день в 17:00.
function advance(ts: number, now: number) {
  const today17 = new Date(now); today17.setUTCHours(17, 0, 0, 0);
  const today19 = new Date(now); today19.setUTCHours(19, 0, 0, 0);
  let t = ts;
  if (t < today17.getTime()) t = today17.getTime();
  if (t <= now) {
    if (t === today17.getTime() && today19.getTime() > now) t = today19.getTime();
    else {
      const d = new Date(now);
      d.setUTCDate(d.getUTCDate() + 1);
      d.setUTCHours(17, 0, 0, 0);
      t = d.getTime();
    }
  }
  return t;
}

Deno.serve(async () => {
  const { data: plants, error } = await sb.from("plants").select("*");
  if (error || !plants) return json({ error: error?.message }, 500);

  const now = Date.now();
  const month = new Date().getMonth() + 1;
  const dueByUser: Record<string, { lines: string[]; patches: { id: string; patch: any }[] }> = {};

  for (const p of plants) {
    const ov = OVR[p.sid];
    const fa = p.feed_from ?? ov?.[0] ?? 3;
    const fb = p.feed_to ?? ov?.[1] ?? 10;
    const feedInSeason = fa <= fb ? (month >= fa && month <= fb) : (month >= fa || month <= fb);

    const jobs: [string, string, number][] = [
      ["💧 Полить", "next_water_at", Date.parse(p.next_water_at)],
      ["🪴 Пересадить", "next_repot_at", Date.parse(p.next_repot_at)],
    ];
    if (feedInSeason) jobs.push(["🧪 Подкормить", "next_feed_at", Date.parse(p.next_feed_at)]);

    const due = jobs.filter(([, , d]) => !isNaN(d) && d <= now);
    if (!due.length) continue;

    const bucket = (dueByUser[p.user_id] ||= { lines: [], patches: [] });
    const patch: any = {};
    due.forEach(([label, col, d]) => {
      bucket.lines.push(`${label}: ${p.emoji} ${p.nickname || p.name}${p.room ? ` · ${p.room}` : ""}`);
      patch[col] = new Date(advance(d, now)).toISOString();
    });
    bucket.patches.push({ id: p.id, patch });
  }

  let sent = 0;
  for (const [uid, b] of Object.entries(dueByUser)) {
    const { data: prof } = await sb.from("profiles")
      .select("chat_id").eq("id", uid).maybeSingle();
    if (!prof?.chat_id) continue;
    const text = `🌿 Green Friend напоминает!\n🗓 ${new Date().toLocaleDateString("ru-RU", {
      weekday: "long", day: "numeric", month: "long",
    })}\n\n${b.lines.join("\n")}\n\nХорошего дня вашим зелёным! 🪴`;
    const r = await fetch(`https://api.telegram.org/bot${BOT}/sendMessage`, {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ chat_id: prof.chat_id, text }),
    });
    if (r.ok) {
      sent++;
      for (const { id, patch } of b.patches) {
        await sb.from("plants").update(patch).eq("id", id);
      }
    }
  }
  return json({ users: Object.keys(dueByUser).length, sent });
});
